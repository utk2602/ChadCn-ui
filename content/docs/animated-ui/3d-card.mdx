## Installation

```bash
npm install framer-motion
```

## Usage

```tsx
import { ThreeDCard } from '@/components/animated-ui/3d-card';

export default function App() {
  return (
    <ThreeDCard>
      <h2 className="text-2xl font-bold">Hover Me</h2>
    </ThreeDCard>
  );
}
```
